@PostMapping("/reviews/{id}/view")
public ResponseEntity<Void> view(@PathVariable Long id, Viewer v) {

    // 같은 사람·같은 글은 30분 버킷당 1회로 센다 — SET key 1 NX EX 1800
    var key = "view:%d:%s:%d".formatted(id, v.id(), epochMin() / 30);
    if (!redis.setIfAbsent(key, "1", ofMinutes(30)))
        return noContent().build();        // 이미 센 조회 → 무시

    // 발행만 하고 바로 리턴 — MySQL은 건드리지 않는다
    kafka.send("review.view.v1",
        new ViewEvent(randomUUID(), id, v.id(), now()));   // event_id 로 뒷단 중복 제거

    return noContent().build();            // 204 · 읽기 경로에 DB 쓰기 0
}
