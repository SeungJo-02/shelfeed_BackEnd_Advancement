// useViewTracker.ts — 본문이 보이는 상태로 3초를 넘겼을 때만 "봤다"
export function useViewTracker(reviewId: number, el: HTMLElement) {
  let t: ReturnType<typeof setTimeout> | undefined;

  const io = new IntersectionObserver(([e]) => {
    if (!e.isIntersecting) { clearTimeout(t); return; }   // 화면에서 벗어나면 취소
    t = setTimeout(() => {
      navigator.sendBeacon(`/api/v1/reviews/${reviewId}/view`);   // 탭을 닫아도 전송된다
      io.disconnect();                                        // 한 화면에서 1회
    }, 3000);                                                // 3초 체류
  }, { threshold: 0.5 });                                     // 본문 절반 이상 노출

  io.observe(el);
  return () => { clearTimeout(t); io.disconnect(); };
}
