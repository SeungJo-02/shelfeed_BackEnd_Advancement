CREATE TABLE view_queue (event_id UUID, review_id UInt64, viewer_id String, ts DateTime)
ENGINE = Kafka SETTINGS kafka_topic_list='review.view.v1', kafka_group_name='ch_view_sink', kafka_format='JSONEachRow';

CREATE TABLE view_stats (review_id UInt64, hour DateTime,
       views AggregateFunction(uniq, UUID), viewers AggregateFunction(uniq, String))
ENGINE = AggregatingMergeTree ORDER BY (review_id, hour);

CREATE MATERIALIZED VIEW view_mv TO view_stats AS
SELECT review_id, toStartOfHour(ts) AS hour,
       uniqState(event_id)  AS views,    -- event_id 기준 근사 집계(중복 무시)
       uniqState(viewer_id) AS viewers   -- 순 방문자
FROM view_queue GROUP BY review_id, hour;

-- 읽기는 ClickHouse를 직접 치지 않는다. 스케줄러가 1분마다 uniqMerge 결과를 Redis에 MSET
